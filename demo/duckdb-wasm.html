timeseries.js — DuckDB-WASM adapter

SQL queries against an in-browser DuckDB, re-run on every viewport change

The DuckDB-WASM adapter needs an AsyncDuckDB instance you bring yourself — there is no synthetic fallback to stub, and the DuckDB-WASM bundle itself is several megabytes so the demo page does not pull it in. Provide the instance via db, or hand the adapter a dbFactory that returns one on first use. 

import TimeSeries from '@hgruber/timeseries.js';
import '@hgruber/timeseries.js/src/adapters/duckdb-wasm.js';
import { attachTooltip } from '@hgruber/timeseries.js/src/tooltip.js';

const ts = new TimeSeries({
  versionMark: true,
  canvas: 'chart',
  sources: [{
    'source-type': 'duckdb-wasm',
    dbFactory: () => initDuckDb(),            // returns a connected AsyncDuckDB
    query: `
      SELECT epoch_ms(time_bucket(INTERVAL (CAST(:mspp AS INT) || ' ms'), ts)) AS t,
             avg(cpu) AS cpu, avg(mem) AS mem
      FROM   metrics
      WHERE  ts BETWEEN to_timestamp(:tmin / 1000.0)
                    AND to_timestamp(:tmax / 1000.0)
      GROUP BY 1
      ORDER BY 1
    `,
    transform(rows) {
      var data = rows.map(r => ({
        t: r.t,
        values: { cpu: r.cpu, mem: r.mem }
      }));
      var vals = data.flatMap(p => Object.values(p.values));
      return {
        category: 'point', type: 'multiline',
        tmin: data[0]?.t ?? 0, tmax: data[data.length - 1]?.t ?? 0,
        min: Math.min(...vals), max: Math.max(...vals),
        series: [{ id: 'cpu', name: 'CPU %' }, { id: 'mem', name: 'Mem %' }],
        data
      };
    }
  }]
});

attachTooltip(ts);

Three placeholders are substituted per query: :tmin and :tmax are the viewport edges in milliseconds; :mspp is 1 / ppms, the milliseconds per pixel — use it for your GROUP BY bucket size so the result is exactly one row per pixel. A query runs once on construction and again on every viewport change.

The transform function receives Arrow rows as a plain array of { column: value } objects and must return a valid BinnedSeries or PointSeries plot object — see data-formats.md.
